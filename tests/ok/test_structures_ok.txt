program {
  real x;
  real y;
  real z;
  real limit;
  bool ok;

  x = 10;
  y = 2;
  limit = 3;

  z = (x + y) * 3 - y / 2;
  ok = (x > y) && !(z < x);

  if (ok) {
    x = x - 1;
  } else {
    x = x + 1;
  }

  if (x < z) {
    y = y + 5;
    ok = true;
  } else {
    y = y - 1;
    ok = false;
  }

  while (x > y) {
    x = x - 1;
    ok = (x >= y) || false;
  }

  while (limit > 0) {
    limit = limit - 1;
  }
}
